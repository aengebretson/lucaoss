find_package(Threads REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/contracts/mcp/foundations-v1.tools.json")
file(READ "${PROJECT_SOURCE_DIR}/contracts/mcp/foundations-v1.tools.json" LUCA_FOUNDATIONS_SCHEMAS)
configure_file(src/schemas.hpp.in generated/schemas.hpp @ONLY)
add_library(luca_application_sqlite STATIC src/application.cpp src/projections.cpp src/reporting.cpp)
add_library(luca::application_sqlite ALIAS luca_application_sqlite)
set_target_properties(luca_application_sqlite PROPERTIES EXPORT_NAME application_sqlite)
target_compile_features(luca_application_sqlite PUBLIC cxx_std_23)
target_include_directories(luca_application_sqlite PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(luca_application_sqlite PUBLIC luca::ingestion_sqlite PRIVATE Threads::Threads)
install(TARGETS luca_application_sqlite EXPORT LucaApplicationSqliteTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.hpp")
install(EXPORT LucaApplicationSqliteTargets FILE LucaApplicationSqliteTargets.cmake NAMESPACE luca:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Luca)
if(LUCA_BUILD_TOOLS)
  add_executable(luca ${PROJECT_SOURCE_DIR}/apps/cli/main.cpp)
  add_executable(luca-mcp ${PROJECT_SOURCE_DIR}/apps/mcp/main.cpp)
  foreach(adapter luca luca-mcp)
    target_link_libraries(${adapter} PRIVATE luca::application_sqlite)
    target_include_directories(${adapter} PRIVATE "${PROJECT_SOURCE_DIR}/apps")
    set_target_properties(${adapter} PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
  endforeach()
  install(TARGETS luca luca-mcp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
